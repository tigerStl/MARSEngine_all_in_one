### 步骤2: 添加运行时保护（10分钟）

在 `App.xaml.cs` 的 `Application_Startup` 方法开头添加：

```csharp
using MarsSpyTool.Utility.License;

private void Application_Startup(object sender, StartupEventArgs e)
{
    logger.Info("Application_Startup\tBegin with security checks");
    
    try
    {
        // ===== 安全检查开始 =====
        
        // 1. 启动反调试保护
        AntiDebugProtection.StartProtection();
        
        // 2. 验证程序集完整性
        if (!AntiDebugProtection.VerifyIntegrity())
        {
            logger.Error("Application_Startup\tIntegrity check failed");
            MessageBox.Show(
                "程序文件已被修改或损坏。\n\n" +
                "请从官方渠道重新下载。\n\n" +
                "如果问题持续，请联系技术支持。",
                "安全警告",
                MessageBoxButton.OK,
                MessageBoxImage.Error);
            Application.Current.Shutdown();
            return;
        }
        
        // ===== 安全检查结束 =====
        
        // 3. License 验证（原有代码）
        string errorMessage = "";
        MarsLicenseManager licenseManager = MarsLicenseManager.Instance;
        
        if (!licenseManager.ValidateLicense(ref errorMessage))
        {
            // ... License 激活流程
        }
        
        // 4. 继续正常启动
        // ... 原有启动代码
    }
    catch (Exception ex)
    {
        logger.Error(ex, $"Application_Startup\tException: {ex.Message}");
        MessageBox.Show(
            "程序启动失败。\n\n" +
            $"错误信息: {ex.Message}",
            "启动错误",
            MessageBoxButton.OK,
            MessageBoxImage.Error);
        Application.Current.Shutdown();
    }
}
```

## 🔧 项目文件配置示例

### MarsSpyTool.csproj

```xml
<?xml version="1.0" encoding="utf-8"?>
<Project ToolsVersion="15.0" xmlns="http://schemas.microsoft.com/developer/msbuild/2003">
  
  <PropertyGroup>
    <!-- 基本配置 -->
    <Configuration Condition=" '$(Configuration)' == '' ">Debug</Configuration>
    <Platform Condition=" '$(Platform)' == '' ">AnyCPU</Platform>
    <OutputType>WinExe</OutputType>
    <TargetFrameworkVersion>v4.7.2</TargetFrameworkVersion>
    
    <!-- 强名称签名配置 -->
    <SignAssembly>true</SignAssembly>
    <AssemblyOriginatorKeyFile>MarsSpyTool.snk</AssemblyOriginatorKeyFile>
    <DelaySign>false</DelaySign>
    
    <!-- 应用程序图标 -->
    <ApplicationIcon>eyeMars.ico</ApplicationIcon>
  </PropertyGroup>
  
  <!-- Release 配置 -->
  <PropertyGroup Condition=" '$(Configuration)|$(Platform)' == 'Release|AnyCPU' ">
    <DebugType>pdbonly</DebugType>
    <Optimize>true</Optimize>
    <OutputPath>bin\Release\</OutputPath>
    <DefineConstants>TRACE</DefineConstants>
    
    <!-- Release 版本也启用签名 -->
    <SignAssembly>true</SignAssembly>
    <AssemblyOriginatorKeyFile>MarsSpyTool.snk</AssemblyOriginatorKeyFile>
  </PropertyGroup>
  
  <!-- 编译后事件：运行混淆（可选） -->
  <PropertyGroup>
    <PostBuildEvent Condition=" '$(Configuration)' == 'Release' ">
      REM 可以在这里调用混淆工具
      REM "$(SolutionDir)Tools\ConfuserEx\Confuser.CLI.exe" "$(SolutionDir)Utility\License\confuser.crproj"
    </PostBuildEvent>
  </PropertyGroup>
  
  <!-- 嵌入式资源 -->
  <ItemGroup>
    <EmbeddedResource Include="Resources\embedded_license.lic" Condition="Exists('Resources\embedded_license.lic')" />
  </ItemGroup>
  
  <!-- License 管理模块 -->
  <ItemGroup>
    <Compile Include="Utility\License\MarsLicenseInfo.cs" />
    <Compile Include="Utility\License\MarsLicenseManager.cs" />
    <Compile Include="Utility\License\LicenseKeyGenerator.cs" />
    <Compile Include="Utility\License\EmbeddedLicenseManager.cs" />
    <Compile Include="Utility\License\AntiDebugProtection.cs" />
    <Compile Include="Utility\License\LicenseActivationWindow.xaml.cs">
      <DependentUpon>LicenseActivationWindow.xaml</DependentUpon>
    </Compile>
  </ItemGroup>
  
  <ItemGroup>
    <Page Include="Utility\License\LicenseActivationWindow.xaml">
      <SubType>Designer</SubType>
      <Generator>MSBuild:Compile</Generator>
    </Page>
  </ItemGroup>
  
  <!-- 必需的引用 -->
  <ItemGroup>
    <Reference Include="System.Management" />
    <Reference Include="Newtonsoft.Json" />
  </ItemGroup>
  
</Project>
```

## 🎯 推荐的构建和发布流程

### 开发阶段

```
1. 编写代码
2. Debug 配置编译（不签名、不混淆）
3. 测试功能
```

### 发布阶段

```bash
# 1. 清理
msbuild /t:Clean

# 2. Release 编译（带签名）
msbuild /p:Configuration=Release

# 3. 运行混淆
.\build_with_obfuscation.ps1

# 4. 测试混淆后的版本
.\test_obfuscation.ps1

# 5. 验证签名
sn -v bin\Release_Obfuscated\MarsSpyTool.exe

# 6. 功能测试
cd bin\Release_Obfuscated
.\MarsSpyTool.exe

# 7. 打包发布
$timestamp = Get-Date -Format "yyyyMMdd"
Compress-Archive -Path "bin\Release_Obfuscated\*" `
    -DestinationPath "Releases\MarsSpyTool_v1.0_${timestamp}.zip"
```

### 自动化脚本

创建 `release.ps1`：

```powershell
param(
    [string]$Version = "1.0.0",
    [switch]$SkipTests
)

Write-Host "═══════════════════════════════════════" -ForegroundColor Cyan
Write-Host "  MARS Spy Tool 发布构建" -ForegroundColor Cyan
Write-Host "  版本: $Version" -ForegroundColor Cyan
Write-Host "═══════════════════════════════════════" -ForegroundColor Cyan

# 1. 清理
Write-Host ""
Write-Host "步骤 1: 清理..." -ForegroundColor Yellow
msbuild /t:Clean /p:Configuration=Release /v:minimal

# 2. 编译
Write-Host ""
Write-Host "步骤 2: 编译..." -ForegroundColor Yellow
msbuild /p:Configuration=Release /v:minimal
if ($LASTEXITCODE -ne 0) { exit 1 }

# 3. 验证签名
Write-Host ""
Write-Host "步骤 3: 验证签名..." -ForegroundColor Yellow
$snPath = "C:\Program Files (x86)\Microsoft SDKs\Windows\v10.0A\bin\NETFX 4.8 Tools\sn.exe"
if (Test-Path $snPath) {
    & $snPath -v "bin\Release\MarsSpyTool.exe"
}

# 4. 混淆
Write-Host ""
Write-Host "步骤 4: 代码混淆..." -ForegroundColor Yellow
.\build_with_obfuscation.ps1

# 5. 测试（可选）
if (-not $SkipTests) {
    Write-Host ""
    Write-Host "步骤 5: 功能测试..." -ForegroundColor Yellow
    # 这里添加自动化测试
    Write-Host "  提示: 请手动测试基本功能" -ForegroundColor Cyan
    Read-Host "  按回车继续"
}

# 6. 打包
Write-Host ""
Write-Host "步骤 6: 打包发布..." -ForegroundColor Yellow
$timestamp = Get-Date -Format "yyyyMMdd_HHmmss"
$packageName = "MarsSpyTool_v${Version}_${timestamp}.zip"
$releasesDir = "Releases"

if (-not (Test-Path $releasesDir)) {
    New-Item -ItemType Directory -Path $releasesDir | Out-Null
}

Compress-Archive -Path "bin\Release_Obfuscated\*" `
    -DestinationPath "$releasesDir\$packageName" `
    -Force

Write-Host ""
Write-Host "═══════════════════════════════════════" -ForegroundColor Green
Write-Host "  发布构建完成!" -ForegroundColor Green
Write-Host "═══════════════════════════════════════" -ForegroundColor Green
Write-Host ""
Write-Host "发布包: $releasesDir\$packageName" -ForegroundColor Cyan
Write-Host "文件大小: $((Get-Item "$releasesDir\$packageName").Length / 1MB | ForEach-Object { '{0:N2}' -f $_ }) MB" -ForegroundColor White
Write-Host ""
```

使用方法：
```powershell
# 标准发布
.\release.ps1 -Version "1.0.0"

# 跳过测试
.\release.ps1 -Version "1.0.1" -SkipTests
```

## ⚠️ 重要注意事项

### 1. 不要过度依赖技术保护

```
技术保护（代码混淆、加密等）
  + 
法律保护（EULA、版权声明）
  + 
商业策略（合理定价、良好服务）
  = 
完整的保护方案
```

### 2. 定期更新保护策略

- 每次发布使用不同的混淆配置
- 定期更新反调试检测方法
- 关注新的破解工具和技术

### 3. 平衡安全性和用户体验

- 不要让安全检查影响正常用户
- 误判时提供友好的解决方案
- 提供技术支持渠道

### 4. 备份和版本管理

```bash
# 始终保留未混淆的版本（用于调试）
bin/Release_Original/       # 原始版本
bin/Release_Obfuscated/     # 混淆版本（发布）

# 版本号管理
MarsSpyTool_v1.0.0_20250109.zip    # 包含日期
```
