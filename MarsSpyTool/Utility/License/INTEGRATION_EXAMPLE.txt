1. 修改 App.xaml.cs - 启动时验证 License
───────────────────────────────────────────────────────────────

在 Application_Startup 方法的开始处添加：

```csharp
using MarsSpyTool.Utility.License;

private void Application_Startup(object sender, StartupEventArgs e)
{
    // ========== License 验证开始 ==========
    logger.Info("Application_Startup\tValidating license...");
    
    string errorMessage = "";
    MarsLicenseManager licenseManager = MarsLicenseManager.Instance;
    
    if (!licenseManager.ValidateLicense(ref errorMessage))
    {
        logger.Warn($"Application_Startup\tLicense invalid: {errorMessage}");
        
        // 显示激活窗口
        LicenseActivationWindow licenseWindow = new LicenseActivationWindow();
        licenseWindow.ShowDialog();
        
        // 再次验证
        if (!licenseManager.ValidateLicense(ref errorMessage))
        {
            logger.Error("Application_Startup\tNo valid license found, shutting down");
            MessageBox.Show(
                "未检测到有效的 License。\n\n" +
                "MARS Spy Tool 需要有效的 License 才能运行。\n" +
                "请联系销售人员获取 License Key。\n\n" +
                "销售联系方式：\n" +
                "Email: sales@mars.com\n" +
                "电话: 400-xxx-xxxx",
                "License 验证失败",
                MessageBoxButton.OK,
                MessageBoxImage.Stop);
            Application.Current.Shutdown();
            return;
        }
    }
    
    // License 有效，检查是否即将过期
    var license = licenseManager.CurrentLicense;
    logger.Info($"Application_Startup\tLicense valid. Type: {license.Type}, Days remaining: {license.GetRemainingDays()}");
    
    if (license.IsExpiringSoon())
    {
        MessageBox.Show(
            $"提醒：您的 {GetLicenseTypeName(license.Type)} License 即将在 {license.GetRemainingDays()} 天后过期。\n\n" +
            $"过期日期：{license.ExpirationDate:yyyy-MM-dd}\n\n" +
            "请及时联系销售续费以确保正常使用。",
            "License 即将过期",
            MessageBoxButton.OK,
            MessageBoxImage.Warning);
    }
    // ========== License 验证结束 ==========
    
    // 以下是原有的启动代码
    string strP = string.Join("|", e.Args);
    try
    {
        // ... 原有代码继续 ...
    }
}

private string GetLicenseTypeName(LicenseType type)
{
    switch (type)
    {
        case LicenseType.Trial: return "试用版";
        case LicenseType.Standard: return "标准版";
        case LicenseType.Professional: return "专业版";
        case LicenseType.Enterprise: return "企业版";
        case LicenseType.Perpetual: return "永久版";
        default: return "未知";
    }
}
```

3. 修改 MainWindow.xaml.cs - 添加菜单事件处理
───────────────────────────────────────────────────────────────

在 MainWindow.xaml.cs 类中添加以下方法：

```csharp
using MarsSpyTool.Utility.License;

// License 管理菜单
private void MenuLicenseManagement_Click(object sender, RoutedEventArgs e)
{
    LicenseActivationWindow licenseWindow = new LicenseActivationWindow();
    licenseWindow.ShowDialog();
}

// License 信息菜单
private void MenuLicenseInfo_Click(object sender, RoutedEventArgs e)
{
    var license = MarsLicenseManager.Instance.CurrentLicense;
    if (license != null && license.IsValid())
    {
        string features = GetFeaturesList(license.Features);
        MessageBox.Show(
            $"═════════ MARS License 信息 ═════════\n\n" +
            $"授权给：{license.LicensedTo}\n" +
            $"License 类型：{GetLicenseTypeName(license.Type)}\n" +
            $"激活状态：{(license.IsActivated ? "已激活" : "未激活")}\n" +
            $"激活日期：{license.ActivationDate:yyyy-MM-dd HH:mm:ss}\n" +
            $"过期日期：{(license.Type == LicenseType.Perpetual ? "永久" : license.ExpirationDate.ToString("yyyy-MM-dd"))}\n" +
            $"剩余天数：{(license.Type == LicenseType.Perpetual ? "永久" : license.GetRemainingDays().ToString() + " 天")}\n" +
            $"硬件ID：{MarsLicenseManager.GetHardwareId().Substring(0, 16)}...\n\n" +
            $"功能权限：\n{features}\n\n" +
            $"═════════════════════════════════════",
            "License 信息",
            MessageBoxButton.OK,
            MessageBoxImage.Information);
    }
    else
    {
        MessageBox.Show(
            "未找到有效的 License。\n\n请点击 [帮助] -> [激活 License] 进行激活。",
            "提示",
            MessageBoxButton.OK,
            MessageBoxImage.Warning);
    }
}

// 激活 License 菜单
private void MenuActivateLicense_Click(object sender, RoutedEventArgs e)
{
    LicenseActivationWindow licenseWindow = new LicenseActivationWindow();
    licenseWindow.ShowDialog();
}

// 关于菜单
private void MenuAbout_Click(object sender, RoutedEventArgs e)
{
    MessageBox.Show(
        "MARS Spy Tool\n" +
        "版本：1.0.0\n\n" +
        "自动化测试对象识别工具\n\n" +
        "Copyright © 2025 MARS Team\n" +
        "All rights reserved.",
        "关于 MARS Spy Tool",
        MessageBoxButton.OK,
        MessageBoxImage.Information);
}

private string GetLicenseTypeName(LicenseType type)
{
    switch (type)
    {
        case LicenseType.Trial: return "试用版";
        case LicenseType.Standard: return "标准版";
        case LicenseType.Professional: return "专业版";
        case LicenseType.Enterprise: return "企业版";
        case LicenseType.Perpetual: return "永久版";
        default: return "未知";
    }
}

private string GetFeaturesList(LicenseFeatures features)
{
    var list = new List<string>();
    if ((features & LicenseFeatures.BasicObjectSpy) != 0)
        list.Add("  ✓ 基础对象识别");
    if ((features & LicenseFeatures.SingleObjectMode) != 0)
        list.Add("  ✓ 单对象模式");
    if ((features & LicenseFeatures.AutoGenerateTestCase) != 0)
        list.Add("  ✓ 自动生成测试用例");
    if ((features & LicenseFeatures.RecordReplay) != 0)
        list.Add("  ✓ 录制回放");
    if ((features & LicenseFeatures.MultiDatabase) != 0)
        list.Add("  ✓ 多数据库支持");
    if ((features & LicenseFeatures.AdvancedObjectRecognition) != 0)
        list.Add("  ✓ 高级对象识别");
    if ((features & LicenseFeatures.BatchOperation) != 0)
        list.Add("  ✓ 批量操作");
    if ((features & LicenseFeatures.CloudSync) != 0)
        list.Add("  ✓ 云端同步");
    return string.Join("\n", list);
}
```

6. 生成 License Key 的工具（独立项目）
───────────────────────────────────────────────────────────────

创建一个新的控制台项目 "MarsLicenseGenerator"：

Program.cs:
```csharp
using System;
using MarsSpyTool.Utility.License;

class Program
{
    static void Main(string[] args)
    {
        Console.WriteLine("MARS License Key Generator v1.0");
        Console.WriteLine("================================\n");
        
        Console.Write("客户名称: ");
        string name = Console.ReadLine();
        
        Console.WriteLine("\n选择 License 类型:");
        Console.WriteLine("1 - 试用版 (30天)");
        Console.WriteLine("2 - 标准版 (1年)");
        Console.WriteLine("3 - 专业版 (1年)");
        Console.WriteLine("4 - 企业版 (1年)");
        Console.WriteLine("5 - 永久版");
        Console.Write("\n请选择: ");
        
        int choice = int.Parse(Console.ReadLine());
        string licenseKey = "";
        
        switch (choice)
        {
            case 1:
                licenseKey = LicenseKeyGenerator.GenerateTrialKey(name);
                break;
            case 2:
                licenseKey = LicenseKeyGenerator.GenerateStandardKey(
                    name, DateTime.Now.AddYears(1));
                break;
            case 3:
                licenseKey = LicenseKeyGenerator.GenerateProfessionalKey(
                    name, DateTime.Now.AddYears(1));
                break;
            case 4:
                licenseKey = LicenseKeyGenerator.GenerateEnterpriseKey(
                    name, DateTime.Now.AddYears(1));
                break;
            case 5:
                licenseKey = LicenseKeyGenerator.GeneratePerpetualKey(name);
                break;
        }
        
        Console.WriteLine("\n================================");
        Console.WriteLine("License Key 生成成功！");
        Console.WriteLine("================================");
        Console.WriteLine($"\n{licenseKey}\n");
        Console.WriteLine("请将此 License Key 发送给客户。");
        Console.WriteLine("\n按任意键退出...");
        Console.ReadKey();
    }
}
```
