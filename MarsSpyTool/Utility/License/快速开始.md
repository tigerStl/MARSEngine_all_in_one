## 🏃 3步快速集成

### 步骤 1：在启动时验证 License（App.xaml.cs）

```csharp
using MarsSpyTool.Utility.License;

private void Application_Startup(object sender, StartupEventArgs e)
{
    // 添加 License 验证
    string errorMessage = "";
    if (!MarsLicenseManager.Instance.ValidateLicense(ref errorMessage))
    {
        // 显示激活窗口
        var licenseWindow = new LicenseActivationWindow();
        licenseWindow.ShowDialog();
        
        // 如果仍然无效，退出程序
        if (!MarsLicenseManager.Instance.ValidateLicense(ref errorMessage))
        {
            MessageBox.Show("未找到有效的 License");
            Application.Current.Shutdown();
            return;
        }
    }
    
    // 检查是否即将过期
    var license = MarsLicenseManager.Instance.CurrentLicense;
    if (license.IsExpiringSoon())
    {
        MessageBox.Show($"您的 License 将在 {license.GetRemainingDays()} 天后过期");
    }
    
    // 原有的启动代码...
}
```

### 步骤 3：生成测试 License Key

创建一个简单的控制台程序：

```csharp
using MarsSpyTool.Utility.License;

static void Main()
{
    // 生成试用版 Key
    string trialKey = LicenseKeyGenerator.GenerateTrialKey("测试用户");
    Console.WriteLine($"试用版 Key:\n{trialKey}");
    
    // 生成专业版 Key（1年有效期）
    string proKey = LicenseKeyGenerator.GenerateProfessionalKey(
        "专业用户", 
        DateTime.Now.AddYears(1), 
        maxActivations: 3);
    Console.WriteLine($"\n专业版 Key:\n{proKey}");
}
```
